                @RequestHeader(TENANT_ID_HEADER) UUID tenantId,

                @RequestHeader(USER_ID_HEADER) UUID userId);

    @Parameter(name = "keyFrom", description = "Ключ файла источника")

    @Parameter(name = "keyTo", description = "Ключ файла назначения")

            produces = MediaType.APPLICATION_JSON_VALUE)

    FileRs move(@RequestHeader(value = "source", required = false) String source,

                @Path(min = 3, max = 1000, nullable = false) @RequestParam String keyFrom,

                @Path(min = 3, max = 1000, nullable = false) @RequestParam String keyTo,

                @RequestHeader(TENANT_ID_HEADER) UUID tenantId,

                @RequestHeader(USER_ID_HEADER) UUID userId);



    @Operation(summary = "Переименования файла")

    @ApiResponse(responseCode = "200", description = "Файл переименован")

    @Parameter(name = "key", description = "Ключ файла")

    @Parameter(name = "fileName", description = "Новое название файла")

                  @NotNull @Pattern(regexp = REGEX) @RequestParam String fileName,

                  @RequestHeader(TENANT_ID_HEADER) UUID tenantId,

                  @RequestHeader(USER_ID_HEADER) UUID userId);

    @Schema(title = "Теги", description = "Дополнительная информация")
